-- 1. USERS TABLE
CREATE TABLE users (
  user_id SERIAL PRIMARY KEY,
  full_name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  password TEXT NOT NULL,
  role TEXT CHECK (role IN ('student', 'teacher', 'admin')) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. EXAMINATION_DATA TABLE
CREATE TABLE examination_data (
  exam_id SERIAL PRIMARY KEY,
  user_id INT REFERENCES users(user_id),
  title TEXT NOT NULL,
  schedule DATE,
  status TEXT DEFAULT 'pending',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. EXAMINATION_ITEM TABLE
CREATE TABLE examination_item (
  exam_item_id SERIAL PRIMARY KEY,
  exam_id INT REFERENCES examination_data(exam_id),
  user_id INT REFERENCES users(user_id),
  
  question_type TEXT CHECK (question_type IN ('multiplechoice', 'truefalse', 'identification')) NOT NULL,
  question_text TEXT NOT NULL,

  option_a TEXT,
  option_b TEXT,
  option_c TEXT,
  option_d TEXT,

  correct_answer TEXT NOT NULL,

  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
